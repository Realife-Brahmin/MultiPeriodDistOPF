\section{Copper-Plate Storage-Scheduling Model}
\label{sec:copper_plate}

\subsection{Optimization Problem}

A single storage device and a single grid connection serve a known demand $p_L^t$
over $T$ periods of length $\Delta t$, all quantities in per unit.  Following the
sign convention of~\cite{Jha2025}, $P_B^t > 0$ discharges and $P_B^t < 0$ charges;
$B^t$ is the energy at the \emph{start} of period $t$, so $B^1$ is given and
$B^{T+1}$ is terminal.  With $t \in \mathcal{T} = \{1,\dots,T\}$:
\begin{subequations}
\label{eq:cp_all}
\begin{alignat}{2}
\min_{P_{\text{Subs}},\,P_B,\,B} \;\;
  & \sum_{t=1}^{T} \Bigl[ c^t P_{\text{Subs}}^t
                        + C_B \bigl(P_B^t\bigr)^2 \Bigr] \Delta t \nonumber \\[-2pt]
  & \hspace{34pt} +\; w \bigl(B^{T+1} - B^1\bigr)^2
\label{eq:cp_obj} \\[2pt]
\text{s.t.} \;\;
  & P_{\text{Subs}}^t + P_B^t - p_L^t = 0, &\quad& t \in \mathcal{T}
\label{eq:cp_balance} \\
  & B^1 = B_0
\label{eq:cp_soc_init} \\
  & B^{t+1} = B^t - P_B^t \Delta t, && t \in \mathcal{T}
\label{eq:cp_soc_dyn} \\
  & P_{\text{Subs}}^t \geq 0, && t \in \mathcal{T}
\label{eq:cp_pg_lim} \\
  & P_B^{\min} \leq P_B^t \leq P_B^{\max}, && t \in \mathcal{T}
\label{eq:cp_pb_lim} \\
  & B^{\min} \leq B^{t+1} \leq B^{\max}, && t \in \mathcal{T}
\label{eq:cp_soc_lim}
\end{alignat}
\end{subequations}
The \emph{base instance} is
\eqref{eq:cp_obj}--\eqref{eq:cp_soc_dyn}, with all three bounds dropped.

\subsection{Optimal-Control Form}
\label{subsec:ocp}

Restated over $N = T$ stages in the form required by DDP:
\begin{align}
\min_{x,u} \;\; & \sum_{t=1}^{N-1} \ell\bigl(x^t, u^t\bigr)
                  + \ell^{F}\bigl(x^N, u^N\bigr) \nonumber \\
\text{s.t.} \;\; & x^1 = \hat{x}^1, \label{eq:ocp_form} \\
                 & x^{t+1} = f\bigl(x^t, u^t\bigr),
                   \quad t \in \{1,\dots,N-1\}, \nonumber \\
                 & c\bigl(x^t, u^t\bigr) = 0,
                   \quad t \in \{1,\dots,N\}. \nonumber
\end{align}
with state, control and initial condition
\begin{equation}
x^t = \bigl(B^t,\; \tau^t\bigr)^\top \in \mathbb{R}^2,
\qquad
u^t = \bigl(P_{\text{Subs}}^t,\; P_B^t\bigr)^\top \in \mathbb{R}^2 ,
\label{eq:ocp_xu}
\end{equation}
$\hat{x}^1 = (B_0,\, 1)^\top$.  Period-varying data is carried by the time index
$\tau$ through the Lagrange interpolants on the nodes $\tau = 1,\dots,T$,
\begin{equation}
p_L(\tau) = \sum_{k=1}^{T} p_L^k \prod_{\substack{m=1 \\ m \neq k}}^{T}
             \frac{\tau - m}{k - m},
\label{eq:ocp_interp}
\end{equation}
and likewise for the price $c(\tau)$.  The stage data is then
\begin{align}
f(x, u) &= \bigl(\, x_1 - u_2 \Delta t,\;\; x_2 + 1 \,\bigr)^\top,
\label{eq:ocp_f} \\
c(x, u) &= u_1 + u_2 - p_L(x_2),
\label{eq:ocp_c} \\
\ell(x, u) &= \bigl[\, c(x_2)\, u_1 + C_B\, u_2^2 \,\bigr] \Delta t,
\label{eq:ocp_l} \\
\ell^{F}(x, u) &= \bigl[\, c^T u_1 + C_B\, u_2^2 \,\bigr] \Delta t \nonumber \\
               &\qquad + w \bigl( x_1 - u_2 \Delta t - B_0 \bigr)^2 .
\label{eq:ocp_lF}
\end{align}
Bounds \eqref{eq:cp_pg_lim}--\eqref{eq:cp_pb_lim} map onto control limits
$b^{L} \leq u^t \leq b^{U}$.  The state bound~\eqref{eq:cp_soc_lim} is appended as
a second row of $c$ via a slack $s^t$ adjoined to the control, giving $n_c = 2$,
$n_u = 3$:
\begin{equation}
\bigl(x_1 - u_2 \Delta t\bigr) - B^{\min} - s^t = 0,
\qquad 0 \leq s^t \leq B^{\max} - B^{\min}.
\label{eq:ocp_slack}
\end{equation}

\subsection{Closed-Form Reference}

The base instance is a strictly convex equality-constrained QP.  Eliminating
$P_{\text{Subs}}^t = p_L^t - P_B^t$ and
$B^{T+1} - B^1 = -\Delta t \sum_t P_B^t$ reduces it to
\begin{align}
J(P_B) = \text{const} & - \Delta t \sum_t c^t P_B^t
       + C_B \Delta t \sum_t \bigl(P_B^t\bigr)^2 \nonumber \\
       & + w (\Delta t)^2 s^2 ,
\label{eq:cp_reduced}
\end{align}
with $s = \sum_t P_B^t$.  Stationarity divided through by $\Delta t$ is
$-c^k + 2 C_B P_B^k + 2 w \Delta t\, s = 0$; summing over $k$ collapses it to
\begin{equation}
s = \frac{\sum_{t} c^t}{2 C_B + 2 T w \Delta t},
\qquad
P_B^k = \frac{c^k - 2 w \Delta t\, s}{2 C_B} .
\label{eq:cp_closed_form}
\end{equation}
